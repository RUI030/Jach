"""Server endpoints."""

"""Return a fixed reply, to prove the client can reach the server."""
def:pub ping -> str {
    return "pong";
}

"""A task node, connected to root. `title` is the typed task text."""
node Task {
    has title: str = "";
}

"""Fetch every `Task` node connected to root, in insertion order."""
def:pub list_tasks -> list[str] {
    tasks = [root-->];
    return [t.title for t in tasks if isinstance(t, Task)];
}

"""Create a new `Task` node connected to root with the given title."""
def:pub add_task(title: str) {
    root ++> Task(title=title);
}

"""Delete the task with the given title, if it exists."""
def:pub delete_task(title: str) {
    for t in [root-->] {
        if isinstance(t, Task) and t.title == title {
            del t;
            return;
        }
    }
}

"""Client UI."""

import from server { ping, list_tasks, add_task, delete_task }

"""Show the page: a task input, an Add button, and the stored tasks."""
def:pub app -> JsxElement {
    has tasks: list[str] = [];
    has new_title: str = "";

    async def onAdd {
        if new_title.strip() == "" {
            return;
        }
        add_task(new_title.strip());
        new_title = "";
        tasks = await list_tasks();
    }

    async def onClear {
        for t in [root-->] {
            del t;
        }
        tasks = await list_tasks();
    }

    async def onRemove(title: str) {
        delete_task(title);
        tasks = await list_tasks();
    }

    async def onClearAll {
        onClear();
    }

    async def onRefresh {
        tasks = await list_tasks();
    }

    async def onPing {
        reply = await ping();
    }

    <div>
        <h1>To-do list</h1>
        <div>
            <input
                type="text"
                placeholder="New task"
                value={new_title}
                onChange={lambda (e: ChangeEvent) { new_title = e.target.value; }}
            />
            <button onClick={onAdd}>Add</button>
        </div>
        <ul>
            {for t in tasks {
                <li key={t}>
                    <span>{t}</span>
                    <button onClick={lambda { onRemove(t); }}>x</button>
                </li>
            }}
        </ul>
        <div>
            <button onClick={onClear}>Clear</button>
            <button onClick={onRefresh}>Refresh</button>
            <button onClick={onClearAll}>Clear all</button>
        </div>
        <button onClick={onPing}>Ping</button>
    </div>
}
